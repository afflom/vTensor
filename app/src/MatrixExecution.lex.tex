  name: evaluate,
  parameters: [

    m : Nat,
    k : Nat,

    right : Strides,

  result: Instruction,

(lc + rc)) then (let element : Nat := (prim subtract phase lc : Nat); (ctor Instruction.transfer false 1 phase (address (prim quotient element n 0 : Nat) (prim remainder element n 0 : Nat) i2 i3 right))) else (if (phase =? (lc + rc)) then (ctor Instruction.multiply) else (let element : Nat := (prim subtract (prim subtract phase (lc + rc) : Nat) 1 : Nat); (ctor Instruction.transfer true 2 ((lc + rc) + element) (address (prim quotient element n 0 : Nat) (prim remainder element n 0 : Nat) i2 i3 output))))))))))) else (ctor Instruction.refused)))))))) else (ctor Instruction.refused))),

  production: {effects: [overflow], targets: [rust-core]}

definition{
  name: scalarLoad,
  parameters: [slot : Nat, at : Nat, offset : Nat, source : Nat, dotType : Nat],
  result: Instruction,
  body: (if ((source =? 0) && (dotType =? 0)) then (ctor Instruction.transfer false slot at offset) else (ctor Instruction.loadConverted slot at offset source dotType (if (source =? 0) then 4 else 2))),
  executable: true
}

definition{
  name: evaluateStorage,
  parameters: [
    index : Nat,
    m : Nat,
    k : Nat,
    n : Nat,
    b2 : Nat,
    b3 : Nat,
    r2 : Nat,
    r3 : Nat,
    left : Strides,
    right : Strides,
    output : Strides,
    leftType : Nat,
    rightType : Nat
  ],
  result: Instruction,
  body: (match (evaluate index m k n b2 b3 r2 r3 left right output)
    | Instruction.transfer writing slot at offset => (if writing then (ctor Instruction.transfer true slot at offset) else (scalarLoad slot at offset (if (slot =? 0) then leftType else rightType) (ScalarDotStorage::activationType leftType)))
    | Instruction.schedule steps total => (ctor Instruction.schedule steps total)
    | Instruction.multiply => (ctor Instruction.multiply)
    | Instruction.refused => (ctor Instruction.refused)
    | Instruction.loadConverted slot at offset source dotType width => (ctor Instruction.loadConverted slot at offset source dotType width)),
  executable: true,
  production: {effects: [overflow], targets: [rust-core]}
}
