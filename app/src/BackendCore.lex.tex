definition{
  name: contractComplete,
  parameters: [coverage : BackendContractCoverage],
  result: Bool,
  body: (((project coverage registry) && (project coverage device)) && ((project coverage buffers) && ((project coverage transfers) && ((project coverage synchronization) && ((project coverage graphExecution) && ((project coverage operations) && ((project coverage tensorTypes) && ((project coverage verdantTensorOnly) && (project coverage kernelCatalog)))))))))
}

definition{name: booleanTrue, parameters: [flag : Bool], result: Prop, body: (flag = true)}

definition{
  name: supportsF64Matmul4,
  parameters: [typeIsF64 : Bool, lhsShapeIs4x4 : Bool, rhsShapeIs4x4 : Bool],
  result: Bool,
  body: ((typeIsF64 && lhsShapeIs4x4) && rhsShapeIs4x4)
}

definition{
  name: SupportsF64Matmul4,
  parameters: [typeIsF64 : Bool, lhsShapeIs4x4 : Bool, rhsShapeIs4x4 : Bool],
  result: Prop,
  body: (((booleanTrue typeIsF64) /\ (booleanTrue lhsShapeIs4x4)) /\ (booleanTrue rhsShapeIs4x4))
}

  name: ContractComplete,
  parameters: [coverage : BackendContractCoverage],
  result: Prop,
  body: (((booleanTrue (project coverage registry)) /\ (booleanTrue (project coverage device))) /\ ((booleanTrue (project coverage buffers)) /\ ((booleanTrue (project coverage transfers)) /\ ((booleanTrue (project coverage synchronization)) /\ ((booleanTrue (project coverage graphExecution)) /\ ((booleanTrue (project coverage operations)) /\ ((booleanTrue (project coverage tensorTypes)) /\ ((booleanTrue (project coverage verdantTensorOnly)) /\ (booleanTrue (project coverage kernelCatalog))))))))))
}

  name: contractCompleteDecides,
  parameters: [coverage : BackendContractCoverage],

  axioms: [propext]
}

theorem{
  name: supportsF64Matmul4Decides,

    definitions: [Bool.and_eq_true, SupportsF64Matmul4, booleanTrue, supportsF64Matmul4]
