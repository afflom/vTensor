inductive{
  name: TransferAdmission,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: missingBuffer, fields: []},
    {name: emptyTransfer, fields: []},
    {name: missingData, fields: []},
    {name: outOfBounds, fields: []},
    {name: admitted, fields: []}
  ]
}

definition{
  name: admitTransfer,
  parameters: [hasBuffer : Bool, hasData : Bool, capacity : Nat, offset : Nat, size : Nat],
  result: TransferAdmission,
  body: (if hasBuffer then (if (size =? 0) then (ctor TransferAdmission.emptyTransfer) else (if hasData then (if ((offset <=? capacity) && (size <=? (prim subtract capacity offset : Nat))) then (ctor TransferAdmission.admitted) else (ctor TransferAdmission.outOfBounds)) else (ctor TransferAdmission.missingData))) else (ctor TransferAdmission.missingBuffer)),
  executable: true,
  production: {effects: [], targets: [rust-core]}
}

theorem{
  name: missingBufferAlwaysRefused,
  parameters: [hasData : Bool, capacity : Nat, offset : Nat, size : Nat],
  statement: ((admitTransfer false hasData capacity offset size) = (ctor TransferAdmission.missingBuffer)),
  proof: reflexivity{}
}

theorem{
  name: emptyTransferNeedsNoAllocation,
  parameters: [hasData : Bool, capacity : Nat, offset : Nat],
  statement: ((admitTransfer true hasData capacity offset 0) = (ctor TransferAdmission.emptyTransfer)),
  proof: reflexivity{}
}

inductive{
  name: Region,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: emptyRegion, fields: []},
    {name: invalidTensor, fields: []},
    {name: outsideBuffer, fields: []},
    {name: located, fields: [Nat]}
  ]
}

definition{
  name: locateTransfer,
  parameters: [base : Nat, capacity : Nat, data : Nat, span : Nat, offset : Nat, size : Nat],
  result: Region,
  body: (match (admitTransfer true (0 <? data) span offset size)
    | TransferAdmission.emptyTransfer => (ctor Region.emptyRegion)
    | TransferAdmission.missingBuffer => (ctor Region.invalidTensor)
    | TransferAdmission.missingData => (ctor Region.invalidTensor)
    | TransferAdmission.outOfBounds => (ctor Region.outsideBuffer)
    | TransferAdmission.admitted => (if (data <? base) then (ctor Region.outsideBuffer) else (let displacement : Nat := (prim subtract data base : Nat); (match (admitTransfer true (0 <? base) capacity displacement span)
      | TransferAdmission.admitted => (ctor Region.located (displacement + offset))
      | TransferAdmission.emptyTransfer => (ctor Region.outsideBuffer)
      | TransferAdmission.missingBuffer => (ctor Region.outsideBuffer)
      | TransferAdmission.missingData => (ctor Region.outsideBuffer)
      | TransferAdmission.outOfBounds => (ctor Region.outsideBuffer))))),
  executable: true,
  production: {effects: [overflow], targets: [rust-core]}
}

theorem{
  name: viewTransferFromBackingBase,
  parameters: [],
  statement: ((locateTransfer 100 40 110 20 5 15) = (ctor Region.located 15)),
  proof: reflexivity{}
}

theorem{
  name: tensorMustFitBackingBuffer,
  parameters: [],
  statement: ((locateTransfer 100 15 110 20 0 1) = (ctor Region.outsideBuffer)),
  proof: reflexivity{}
}

theorem{
  name: emptyTransferRequiresNoAddress,
  parameters: [],
  statement: ((locateTransfer 0 0 0 0 100 0) = (ctor Region.emptyRegion)),
  proof: reflexivity{}
}
